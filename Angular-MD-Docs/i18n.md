# Angular i18n Cheat Sheet

## 1. Native `@angular/localize` (compile-time i18n)

### Setup

```bash
ng add @angular/localize
```

### Mark text — templates

```html
<h1 i18n="@@app.welcome.header">Welcome</h1>
<p i18n-title title="Tooltip" i18n>Content</p>
<span i18n="meaning|description@@app.greeting">Hello!</span>

<!-- Plural -->
<span i18n
  >{count, plural, =0 {no items} =1 {one item} other {{{count}} items}}</span
>

<!-- Select -->
<span i18n>{gender, select, male {He} female {She} other {They}}</span>
```

### Mark text — TypeScript

```typescript
import "@angular/localize/init";
const msg = $localize`:@@app.error.notFound:Resource not found`;
```

### `angular.json` — project-level i18n config

```json
"i18n": {
  "sourceLocale": "en-US",
  "locales": {
    "hi": { "translation": "src/locale/messages.hi.json", "baseHref": "/hi/" },
    "fr": { "translation": "src/locale/messages.fr.xlf",  "baseHref": "/fr/" }
  }
}
```

### `angular.json` — extract-i18n builder

```json
"extract-i18n": {
  "builder": "@angular-devkit/build-angular:extract-i18n",
  "options": {
    "buildTarget": "your-app:build",
    "format": "json",
    "outputPath": "src/locale",
    "outFile": "messages.json"
  }
}
```

| format | ext     | notes                              |
| ------ | ------- | ---------------------------------- |
| `xlf`  | `.xlf`  | XLIFF 1.2, wide CAT tool support   |
| `xlf2` | `.xlf`  | XLIFF 2.0, recommended for tooling |
| `xmb`  | `.xmb`  | translations returned as `.xtb`    |
| `json` | `.json` | simple flat key-value              |
| `arb`  | `.arb`  | Flutter-compatible                 |

### `angular.json` — build/serve per locale

```json
"build": {
  "configurations": {
    "hi": { "localize": ["hi"] },
    "fr": { "localize": ["fr"] }
  }
},
"serve": {
  "configurations": {
    "hi": { "buildTarget": "your-app:build:hi" }
  }
}
```

### Commands

```bash
ng extract-i18n                          # uses angular.json config
ng extract-i18n --format=xlf2 --output-path=src/locale --out-file=messages.xlf
ng build --localize                      # builds ALL configured locales
ng build --configuration=hi              # single locale build
ng serve --configuration=hi              # dev serve single locale
```

### Output

```
dist/your-app/en-US/
dist/your-app/hi/
dist/your-app/fr/
```

### Key trait

Compile-time only — switching language = loading a different build, **no runtime toggle**.

## 2. ngx-translate (runtime i18n)

### Install

```bash
npm install @ngx-translate/core @ngx-translate/http-loader
```

### Translation files

```json
// assets/i18n/en.json
{ "HEADER": { "TITLE": "Welcome", "GREETING": "Hello, {{name}}!" } }
```

### Config (standalone)

```typescript
export function HttpLoaderFactory(http: HttpClient) {
  return new TranslateHttpLoader(http, "./assets/i18n/", ".json");
}

providers: [
  provideHttpClient(),
  importProvidersFrom(
    TranslateModule.forRoot({
      defaultLanguage: "en",
      loader: {
        provide: TranslateLoader,
        useFactory: HttpLoaderFactory,
        deps: [HttpClient],
      },
    }),
  ),
];
```

### Set/switch language

```typescript
translate.addLangs(["en", "hi", "fr"]);
translate.setDefaultLang("en");
translate.use("hi");
localStorage.setItem("lang", "hi");
```

### Template usage

```html
<h1>{{ 'HEADER.TITLE' | translate }}</h1>
<p>{{ 'HEADER.GREETING' | translate: { name: userName } }}</p>
<h1 [translate]="'HEADER.TITLE'"></h1>
<input [placeholder]="'HEADER.TITLE' | translate" />
```

### TypeScript usage

```typescript
translate.get('HEADER.GREETING', { name: 'Harish' }).subscribe(res => ...);
translate.instant('HEADER.TITLE');   // sync — only if lang already loaded
translate.onLangChange.subscribe(e => console.log(e.lang));
```

### Lazy-loaded module merge

```typescript
http
  .get(`assets/i18n/feature-x/${translate.currentLang}.json`)
  .subscribe((data) =>
    translate.setTranslation(translate.currentLang, data, true),
  );
```

### Gotchas

- `instant()` returns the raw key if translations aren't loaded — prefer `get()` or the pipe.
- No built-in ICU plural/select — needs `@ngx-translate/i18n-polyfill` or manual `Intl.PluralRules`.
- `.json` keys aren't type-checked — typos fail silently at runtime.

## 3. Date / Number / Currency formatting (ngx-translate does NOT handle this)

### Register locale data (only if using Angular's locale pipes directly)

```typescript
import { registerLocaleData } from "@angular/common";
import localeHi from "@angular/common/locales/hi";
registerLocaleData(localeHi, "hi");
```

### `LOCALE_ID` — bootstrap-only, NOT reactive

```typescript
{ provide: LOCALE_ID, useFactory: () => localStorage.getItem('locale') ?? 'en-US' }
```

Changing this later does nothing — pipes read it once at construction.

### Runtime-reactive: pass locale explicitly to built-in pipes

```html
<p>{{ date | date: 'medium' : undefined : localeService.locale() }}</p>
<p>
  {{ price | currency: 'INR' : 'symbol' : '1.2-2' : localeService.locale() }}
</p>
<p>{{ rating | number: '1.0-2' : localeService.locale() }}</p>
```

### Recommended: `Intl`-based service (no locale-data bundle needed)

```typescript
@Injectable({ providedIn: "root" })
export class LocaleFormatService {
  private _locale = signal("en-US");
  readonly locale = this._locale.asReadonly();

  constructor(private translate: TranslateService) {
    this.translate.onLangChange.subscribe((e) =>
      this._locale.set(
        { en: "en-US", hi: "hi-IN", fr: "fr-FR" }[e.lang] ?? "en-US",
      ),
    );
  }

  formatDate(
    date: Date,
    opts: Intl.DateTimeFormatOptions = { dateStyle: "medium" },
  ) {
    return new Intl.DateTimeFormat(this._locale(), opts).format(date);
  }
  formatCurrency(value: number, currency: string) {
    return new Intl.NumberFormat(this._locale(), {
      style: "currency",
      currency,
    }).format(value);
  }
  formatNumber(value: number, opts: Intl.NumberFormatOptions = {}) {
    return new Intl.NumberFormat(this._locale(), opts).format(value);
  }
  pluralCategory(count: number): Intl.LDMLPluralRule {
    return new Intl.PluralRules(this._locale()).select(count);
  }
}
```

### Selective application — only where needed, everything else untouched

```html
<p>{{ order.id }}</p>
<!-- untouched -->
<p>
  {{ order.total | currency: 'INR' : 'symbol' : '1.2-2' : localeService.locale()
  }}
</p>
<!-- locale-aware -->
```

Custom pipe wrapping the service (`pure: false` required — args don't change, only injected signal does):

```typescript
@Pipe({ name: "appCurrency", standalone: true, pure: false })
export class AppCurrencyPipe implements PipeTransform {
  private svc = inject(LocaleFormatService);
  transform(value: number, currency = "INR") {
    return this.svc.formatCurrency(value, currency);
  }
}
```

### Get locale from Angular

```typescript
inject(LOCALE_ID); // static, bootstrap value
getLocaleCurrencyCode("hi-IN"); // "INR"
getLocaleCurrencySymbol("hi-IN"); // "₹"
getLocaleFirstDayOfWeek("en-US"); // 0 (Sunday)
navigator.language / navigator.languages; // browser/OS locale
Intl.DateTimeFormat().resolvedOptions().locale; // resolved effective locale
translate.currentLang; // ngx-translate active lang
```

## 4. Decision guide

| Need                                            | Use                                                                       |
| ----------------------------------------------- | ------------------------------------------------------------------------- |
| Runtime language switch, no rebuild             | ngx-translate                                                             |
| Smaller bundles, compile-time safety            | `@angular/localize`                                                       |
| Date/number/currency formatting                 | Angular locale pipes (explicit locale param) or `Intl` service            |
| Plurals/ICU with `@angular/localize`            | Native ICU syntax in templates                                            |
| Plurals/ICU with ngx-translate                  | `Intl.PluralRules` + keyed JSON, or polyfill lib                          |
| Format only specific fields, leave rest default | Explicit locale pipe param / custom pipe — don't touch global `LOCALE_ID` |
